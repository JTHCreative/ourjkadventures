# for search engines. everything public may be crawled; the owner's page and the api may not.
# pages that should stay out of search (account, sign-in, thank-you and so on) say "noindex" themselves,
# so they are not blocked here: a crawler has to be able to open a page to see its noindex.
# the three public lists the pages draw themselves from are allowed: without the catalog, google sees an empty
# pup board and calls the page a dead end (a "soft 404"). the longer, more exact rule wins, so the rest of
# /api/ (checkout, accounts, admin, votes, emails) stays blocked.
User-agent: *
Allow: /api/catalog
Allow: /api/reviews
Allow: /api/reels
Disallow: /api/
Disallow: /admin
